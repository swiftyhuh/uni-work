        sem_wait(&sa);
        printf("a\n");
        sem_post(&sb);
    }
    return NULL;
}
void* fb(void* args){
    for(int i=0; i<10; i++) {
        sem_wait(&sb);
        printf("b\n");
        sem_post(&sa);
    }
    return NULL;
}
int main(){
    pthread_t ta, tb;
    sem_init(&sa, 0, );
    sem_init(&sb, 0, );
    pthread_create(&ta, NULL, fa, NULL);
    pthread_create(&tb, NULL, fb, NULL);
    pthread_join(ta, NULL);
    pthread_join(tb, NULL);
    sem_destroy(&sa);
    sem_destroy(&sb);
    return 0;
}

16. Dați valori strict pozitive pentru T, N1, N2 pentru care programul de mai jos se termină corect, și orice apel la pthread_barrier_wait(&b1) se deblochează imediat sau cât mai rapid posibil, fără a aștepta alte fire.
#include <pthread.h>
pthread_barrier_t b1, b2;
void* f1(void* a) {
    pthread_barrier_wait(&b1);
    return NULL;
}
void* f2(void* a) {
    pthread_barrier_wait(&b2);
    return NULL;
}
int main() {
    int T, N1, N2;
    pthread_t t[T][2];
    pthread_barrier_init(&b1, NULL, N1);
    pthread_barrier_init(&b2, NULL, N2);
    for(int i=0; i<T; i++) {
        pthread_create(&t[i][0], NULL, f1, NULL);
        pthread_create(&t[i][1], NULL, f2, NULL);
    }
    for(int i=0; i<T; i++) {
        pthread_join(t[i][0], NULL);
        pthread_join(t[i][1], NULL);
    }
    pthread_barrier_destroy(&b1);
    pthread_barrier_destroy(&b2);
    return 0;
}
